-- ⚠️ 초안 — 그대로 실행 금지. cbmc-app 실제 스키마에 맞춰 수정 필요:
--  1) chapter_id: 이 저장소는 uuid가 아니라 text('새서울')이며 chapters(chapter_id) 참조.
--     → chapter_id text not null default '새서울' references chapters(chapter_id)
--  2) is_admin(chapter_id): 실제 함수는 인자 없는 is_admin() (SECURITY DEFINER).
--     읽기 정책은 is_staff()(운영진+읽기운영진) 분리 패턴 유지 권장.
--  3) created_by: 관리자 식별은 members(id)보다 auth.uid()/profiles 기준이 이 앱 관례.
--  4) 서명자 접근: 스펙은 service_role 서버 라우트인데, 이 앱의 기존 관례는
--     체크인처럼 토큰 게이트 SECURITY DEFINER RPC (anon 키). PDF 합성·private 스토리지는
--     서버 라우트 필요 → SUPABASE_SERVICE_ROLE_KEY를 Vercel에 추가할지 구현 시 결정.
--  5) 마이그레이션 번호: supabase/migrations/0055_sign_module.sql 로 확정본 작성할 것.
--
-- 서명 수집 모듈 마이그레이션
-- 전제: chapters(id), members(id, chapter_id) 테이블 존재. 관리자 판별 함수는 프로젝트 기존 방식(is_admin() 등)에 맞춰 수정.

create table if not exists sign_requests (
  id uuid primary key default gen_random_uuid(),
  chapter_id uuid not null references chapters(id),
  title text not null,
  description text,
  source_pdf_path text not null,
  final_pdf_path text,
  status text not null default 'draft' check (status in ('draft','active','completed','expired','cancelled')),
  expires_at timestamptz,
  created_by uuid references members(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists sign_slots (
  id uuid primary key default gen_random_uuid(),
  request_id uuid not null references sign_requests(id) on delete cascade,
  label text not null,
  page int not null,
  x numeric not null, y numeric not null, w numeric not null, h numeric not null,
  order_no int not null default 0
);

create table if not exists sign_signers (
  id uuid primary key default gen_random_uuid(),
  request_id uuid not null references sign_requests(id) on delete cascade,
  slot_id uuid not null references sign_slots(id) on delete cascade,
  member_id uuid references members(id),
  name text not null,
  phone text,
  token text not null unique,
  status text not null default 'pending' check (status in ('pending','viewed','signed','declined')),
  viewed_at timestamptz,
  signed_at timestamptz,
  signature_path text,
  ip text,
  user_agent text,
  auth_kakao_id text,
  created_at timestamptz not null default now()
);

create table if not exists sign_events (
  id bigserial primary key,
  request_id uuid references sign_requests(id) on delete cascade,
  signer_id uuid references sign_signers(id) on delete set null,
  event text not null,
  meta jsonb,
  created_at timestamptz not null default now()
);

create index on sign_slots(request_id);
create index on sign_signers(request_id);
create index on sign_signers(token);
create index on sign_events(request_id, created_at);

-- RLS
alter table sign_requests enable row level security;
alter table sign_slots    enable row level security;
alter table sign_signers  enable row level security;
alter table sign_events   enable row level security;

-- 관리자만 직접 접근. 서명자 화면은 service_role 키를 쓰는 서버 라우트(/api/sign/[token])로만 접근.
-- is_admin(chapter_id) 는 프로젝트의 기존 관리자 판별 함수로 교체.
create policy sign_requests_admin on sign_requests
  for all using (is_admin(chapter_id)) with check (is_admin(chapter_id));

create policy sign_slots_admin on sign_slots
  for all using (exists (select 1 from sign_requests r where r.id = request_id and is_admin(r.chapter_id)));

create policy sign_signers_admin on sign_signers
  for all using (exists (select 1 from sign_requests r where r.id = request_id and is_admin(r.chapter_id)));

create policy sign_events_admin on sign_events
  for select using (exists (select 1 from sign_requests r where r.id = request_id and is_admin(r.chapter_id)));
